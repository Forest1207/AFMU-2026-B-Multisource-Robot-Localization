\subsection{模型一的建立与求解}\label{sec:model-q1}

\subsubsection{模型一的建立}\label{sec:q1-establishment}

\noindent\textbf{（1）真实轨迹采样模型}

设机器人在统一物理时间轴上的真实二维轨迹为
\(\mathbf r^{\star}(t)=(x^{\star}(t),y^{\star}(t))^{\mathsf T}\)。两种定位方式
仅具有不同的设备时间原点，其观测坐标均来自同一条真实轨迹。记第 \(m\) 种方式的
设备时钟偏移为 \(\delta_m\)，并以方式 1 的时间轴为基准，即
\(\delta_1=0\)、\(\delta_2=\Delta t_c<0\)，则第 \(i\) 个观测满足

\begin{equation}
\mathbf p_i^{(m)}=
\begin{pmatrix}x_i^{(m)}\\y_i^{(m)}\end{pmatrix}
=\mathbf r^{\star}\!\left(t_i^{(m)}+\delta_m\right),
\qquad m=1,2.
\label{eq:q1-true-sampling}
\end{equation}

因此两组定位数据可统一写成真实轨迹在各自采样时刻上的采样集合

\begin{equation}
D_m=\left\{\left(t_i^{(m)},\mathbf p_i^{(m)}\right)\right\}_{i=1}^{n_m}
=\left\{\left(t_i^{(m)},
\mathbf r^{\star}(t_i^{(m)}+\delta_m)\right)\right\}_{i=1}^{n_m}.
\label{eq:q1-sampling-set}
\end{equation}

式\eqref{eq:q1-true-sampling}中不含随机误差项，故时间对齐的核心是辨识
\(\Delta t_c\)，而不是估计或滤除位置噪声。

\noindent\textbf{（2）连续轨迹重建}

由于两种方式的采样时刻不同，先分别根据离散观测构造连续坐标函数。记第 \(m\) 种
方式在第 \(i\) 个采样点的某一坐标分量为 \(z_i^{(m)}\)，其中
\(z\in\{x,y\}\)。在区间 \([t_i^{(m)},t_{i+1}^{(m)}]\) 上构造三次多项式

\begin{equation}
S_{m,z,i}(t)=a_{m,z,i}+b_{m,z,i}\tau+c_{m,z,i}\tau^2+d_{m,z,i}\tau^3,
\qquad \tau=t-t_i^{(m)}.
\label{eq:q1-spline-piece}
\end{equation}

各段满足节点插值条件及二阶光滑连接条件

\begin{equation}
S_{m,z}(t_i^{(m)})=z_i^{(m)},\qquad
S_{m,z,i-1}^{(q)}(t_i^{(m)})=S_{m,z,i}^{(q)}(t_i^{(m)}),
\quad q=0,1,2,
\label{eq:q1-spline-conditions}
\end{equation}

并在两端采用 not-a-knot 条件。由此得到连续二维轨迹

\begin{equation}
\mathbf r_m(t)=\bigl(S_{m,x}(t),S_{m,y}(t)\bigr)^{\mathsf T},
\qquad m=1,2.
\label{eq:q1-continuous-trajectory}
\end{equation}

后续时间对齐与轨迹重建均采用式\eqref{eq:q1-continuous-trajectory}定义的连续轨迹。

\noindent\textbf{（3）时间映射与位置误差目标}

将\emph{加到方式 2 原始时间戳上的校正量}定义为 \(\Delta t_c\)，即

\begin{equation}
t_{2,\mathrm{aligned}}=t_2+\Delta t_c,
\qquad
\Delta t_c:=t_{2,\mathrm{aligned}}-t_2<0.
\label{eq:q1-time-map}
\end{equation}

因此 \(|\Delta t_c|\) 表示方式 2 时间轴向前平移的时长。参考时刻 \(t\) 在
方式 2 原始时间轴上对应 \(t-\Delta t_c\)。对任一候选校正量 \(\delta\)，两条
连续轨迹的公共有效区间为

\begin{equation}
\Omega(\delta)=
\left[
\max(t_{1,\min},t_{2,\min}+\delta),
\min(t_{1,\max},t_{2,\max}+\delta)
\right].
\label{eq:q1-overlap}
\end{equation}

要求公共区间长度不小于给定阈值 \(T_{\min}\)，从而定义可行域

\begin{equation}
\mathcal D=\left\{\delta:\ |\Omega(\delta)|\ge T_{\min}\right\}
=\left[\delta_{\min},\delta_{\max}\right],
\label{eq:q1-feasible-domain}
\end{equation}

其中

\begin{equation}
\delta_{\min}=t_{1,\min}-t_{2,\max}+T_{\min},\qquad
\delta_{\max}=t_{1,\max}-t_{2,\min}-T_{\min}.
\label{eq:q1-feasible-bounds}
\end{equation}

在公共评价时刻 \(\{t_k\}\) 上定义位置均方误差

\begin{equation}
J(\delta)=\frac{1}{N(\delta)}
\sum_{k=1}^{N(\delta)}
\left\|\mathbf r_1(t_k)-\mathbf r_2(t_k-\delta)\right\|_2^2,
\qquad t_k\in\Omega(\delta),
\label{eq:q1-objective}
\end{equation}

从而得到

\begin{equation}
\Delta t_c^*=\underset{\delta\in\mathcal D}{\arg\min}\ J(\delta).
\label{eq:q1-optimum}
\end{equation}

\subsubsection{模型一的求解}\label{sec:q1-solution}

\noindent\textbf{（1）速度特征归一化互相关}

为消除原始采样频率不同造成的数组长度差异，先按

\begin{equation}
u_j=jh_c,\qquad h_c>0,
\label{eq:q1-corr-grid}
\end{equation}

分别在两台设备各自的相对时间轴上重采样。以二阶中心差分近似速度特征

\begin{equation}
\mathbf v_{m,j}=\frac{\mathbf r_m(t_{m,\min}+u_j+h_c)
-\mathbf r_m(t_{m,\min}+u_j-h_c)}{2h_c},
\label{eq:q1-velocity-feature}
\end{equation}

端点处采用相应的二阶单边差分。为使两个方向的速度分量具有可比尺度，对
\(c\in\{x,y\}\) 作标准化

\begin{equation}
z_{m,c,j}=\frac{v_{m,c,j}-\bar v_{m,c}}{s_{m,c}},
\label{eq:q1-standardized-velocity}
\end{equation}

并定义离散滞后 \(\ell\) 下的二维归一化互相关系数

\begin{equation}
\rho(\ell)=\frac{1}{2N_\ell}
\sum_{c\in\{x,y\}}\ \sum_{j\in I_\ell}
z_{1,c,j}\,z_{2,c,j-\ell},
\label{eq:q1-ncc}
\end{equation}

其中 \(I_\ell\) 为两序列在该滞后下的重叠索引集，\(N_\ell=|I_\ell|\)。滞后量
与时间校正候选之间满足

\begin{equation}
\delta_\ell=t_{1,\min}+\ell h_c-t_{2,\min}.
\label{eq:q1-lag-offset}
\end{equation}

仅保留重叠时长不少于 \(T_{\min}\) 的局部相关峰，并按峰间距不少于
\(L_{\min}\) 的条件选取相关系数较大的前 \(K\) 个候选，构成集合
\(\mathcal C\)。互相关最大值不直接作为最终结果，以免周期性运动产生伪峰。

\noindent\textbf{（2）全域扫描与 Brent 精细搜索}

在式\eqref{eq:q1-feasible-domain}给出的全可行域内，以粗搜索步长 \(h_g>0\)
构造网格

\begin{equation}
\mathcal D_g=\left\{\delta_{\min}+jh_g:
j=0,1,\ldots,
\left\lfloor\frac{\delta_{\max}-\delta_{\min}}{h_g}\right\rfloor\right\},
\qquad
\delta_0=\underset{\delta\in\mathcal D_g}{\arg\min}\ J(\delta).
\label{eq:q1-coarse-seed}
\end{equation}

随后给定局部搜索半径 \(w>0\)，在区间

\begin{equation}
[a,b]=\left[\max(\delta_{\min},\delta_0-w),
\min(\delta_{\max},\delta_0+w)\right]
\label{eq:q1-brent-bracket}
\end{equation}

内采用有界 Brent 法精修。为避免
候选校正量改变评价网格而引入数值非光滑，固定所有候选均有效的区间

\begin{equation}
\Omega_B=
\left[
\max(t_{1,\min},t_{2,\min}+b),
\min(t_{1,\max},t_{2,\max}+a)
\right],
\label{eq:q1-brent-overlap}
\end{equation}

并以精细评价步长 \(h_f>0\) 生成固定评价网格。Brent 法在抛物线插值与黄金分割
步之间自适应切换，最终求解

\begin{equation}
\Delta t_c^*=\underset{\delta\in[a,b]}{\arg\min}\
\frac{1}{N_B}\sum_{t_k\in\Omega_B}
\left\|\mathbf r_1(t_k)-\mathbf r_2(t_k-\delta)\right\|_2^2,
\label{eq:q1-brent-objective}
\end{equation}

当搜索区间宽度不超过给定容差 \(\varepsilon_t>0\) 时终止迭代。

\noindent\textbf{（3）同时刻观测合并与10 Hz重建}

对方式 2 时间戳施加 \(\Delta t_c^*\)
后，将两组观测合并、排序。若两个观测落在
同一真实时刻，则先检查坐标差，再按

\begin{equation}
\bar{\mathbf r}(\tau)=\frac{1}{|I_\tau|}
\sum_{(m,i)\in I_\tau}\mathbf r_i^{(m)},
\qquad
I_\tau=\{(m,i):t_{i,\mathrm{aligned}}^{(m)}=\tau\}
\label{eq:q1-merge-duplicate}
\end{equation}

合并为一个节点。对合并后的唯一时刻序列建立三次样条，记其有效时间边界为

\begin{equation}
t_{\min}^{\mathrm{out}}=\min(t_{1,\min},t_{2,\min}+\Delta t_c^*),\qquad
t_{\max}^{\mathrm{out}}=\max(t_{1,\max},t_{2,\max}+\Delta t_c^*).
\label{eq:q1-output-domain}
\end{equation}

令 10 Hz 输出步长为 \(h_o=0.1\ \mathrm{s}\)，并记
\(N_o=\left\lfloor(t_{\max}^{\mathrm{out}}-t_{\min}^{\mathrm{out}})/h_o\right\rfloor\)，
则输出网格为

\begin{equation}
t_n=t_{\min}^{\mathrm{out}}+nh_o,\qquad n=0,1,\ldots,N_o.
\label{eq:q1-grid}
\end{equation}

\subsubsection{算法实现}\label{sec:q1-algorithm}

\begin{pseudocodeblock}{\textbf{Algorithm 1}\quad 无噪声时间配准与10 Hz轨迹重建}
\noindent\textbf{Require:}\quad 两路定位序列 \(D_1,D_2\)，参数 \(T_{\min},h_c,h_g,h_f,h_o,w,\varepsilon_t\)\par
\noindent\textbf{Ensure:}\quad 负时间校正量 \(\Delta t_c^*\) 与10 Hz二维位置轨迹 \(\widehat{\mathbf r}(t_n)\)\par
\medskip
\begin{pseudocodelines}
\pseudostep{按式\eqref{eq:q1-spline-piece}--\eqref{eq:q1-continuous-trajectory}构造连续轨迹 \(\mathbf r_1(t),\mathbf r_2(t)\)}
\pseudostep{在步长为 \(h_c\) 的网格上计算并标准化二维速度特征}
\pseudostep{由式\eqref{eq:q1-ncc}计算互相关峰，生成候选集合 \(\mathcal C\)}
\pseudostep{\textbf{for} 每个候选校正量 \(\delta\in\mathcal C\) \textbf{do}}
\pseudostep[2em]{按式\eqref{eq:q1-overlap}计算公共有效区间 \(\Omega(\delta)\)}
\pseudostep[2em]{按式\eqref{eq:q1-objective}计算位置均方误差 \(J(\delta)\)}
\pseudostep{\textbf{end for}}
\pseudostep{在网格 \(\mathcal D_g\) 上扫描 \(J(\delta)\)，选择最小值点 \(\delta_0\) 作为初值}
\pseudostep{按式\eqref{eq:q1-brent-bracket}构造 \([a,b]\)，用 Brent 法求得 \(\Delta t_c^*\)}
\pseudostep{按式\eqref{eq:q1-time-map}校正方式2时间戳，并按式\eqref{eq:q1-merge-duplicate}合并同刻观测}
\pseudostep{按式\eqref{eq:q1-grid}生成时刻 \(t_n\)，重建轨迹 \(\widehat{\mathbf r}(t_n)\)}
\pseudostep{\textbf{return} \(\Delta t_c^*,\{\widehat{\mathbf r}(t_n)\}_{n=0}^{N_o}\)}
\end{pseudocodelines}
\end{pseudocodeblock}

\subsubsection{结果分析}\label{sec:q1-results}

当公共区间长度阈值取 \(T_{\min}=60\ \mathrm{s}\) 时，式
\eqref{eq:q1-feasible-bounds}对应的时间校正量可行域为

\begin{equation}
\mathcal D=[-987.8317,\,441.9183]\ \mathrm{s}.
\label{eq:q1-numerical-domain}
\end{equation}

速度特征以 \(h_c=0.1\ \mathrm{s}\) 重采样，互相关峰经重叠时长和峰间距约束后
保留 8 个候选。以 \(h_g=0.5\ \mathrm{s}\) 扫描式
\eqref{eq:q1-objective}，位置误差最小的粗搜索点为

\begin{equation}
\delta_0=-198.3317\ \mathrm{s},
\label{eq:q1-numerical-coarse}
\end{equation}

Brent 搜索区间取 \([-199.3317,-197.3317]\ \mathrm{s}\)，固定评价网格步长为
\(h_f=0.05\ \mathrm{s}\)，横坐标终止容差为 \(10^{-10}\ \mathrm{s}\)。连续优化值为

\begin{equation}
\Delta t_c^*=-198.4316999986\ \mathrm{s}.
\label{eq:q1-raw-offset}
\end{equation}

输入时间戳保留 4 位小数，故时间校正量取

\begin{equation}
\boxed{\Delta t_c^*=-198.4317\ \mathrm{s}}.
\label{eq:q1-reported-offset}
\end{equation}

图\ref{fig:q1-objective}同时展示最优邻域内的速度互相关与位置均方误差。互相关峰
与误差谷值均位于 \(-198.43\ \mathrm{s}\) 附近，与式
\eqref{eq:q1-reported-offset}一致。图中按保留两位小数的偏差绘制，因而标注的 RMSE
为 \(6.6\times10^{-8}\ \mathrm{m}\)；采用式\eqref{eq:q1-raw-offset}中的未舍入偏差时，
位置误差为

\begin{equation}
\operatorname{RMSE}=\sqrt{J(\Delta t_c^*)}
=8.51\times10^{-11}\ \mathrm{m}
\label{eq:q1-rmse}
\end{equation}
